## Upgrade steps

```bash
pnpm add -D @kiwa/desktop@^0.6
```

### Pattern 1 — dry-run 経路 (backward compat)

```ts
import { invokeDesktopCli } from '@kiwa/desktop';

const result = await invokeDesktopCli({
  command: 'ffmpeg',
  args: ['-version'],
  env: { KIWA_DESKTOP_MODE: 'real', KIWA_DESKTOP_SPAWN: 'dry-run', PATH: '/usr/bin' },
});
console.log(result.stdout); // "[v0.6 dry-run] ffmpeg -version"
```

### Pattern 2 — DI 経路 (SpawnFn 注入)

```ts
import { invokeDesktopCliWith, type SpawnFn } from '@kiwa/desktop';

const dummySpawn: SpawnFn = /* test-only spawn */;
const result = await invokeDesktopCliWith(
  {
    command: 'ffmpeg',
    args: ['-version'],
    env: { KIWA_DESKTOP_MODE: 'real', PATH: '/usr/bin' },
  },
  dummySpawn,
);
```

### Pattern 3 — sanitizeEnv per-command allowlist

```ts
import { sanitizeEnv } from '@kiwa/desktop';

const clean = sanitizeEnv('electron-builder', {
  PATH: '/usr/bin',
  CSC_LINK: 'ok',
  SECRET_TOKEN: 'nope', // 削ぎ落とし
});
// clean.SECRET_TOKEN === undefined
```

### Pattern 4 — executeSpawn (低レベル API)

```ts
import { executeSpawn, type SpawnExecutorInput } from '@kiwa/desktop';

const input: SpawnExecutorInput = {
  command: 'ffmpeg',
  args: ['-version'],
  env: { PATH: '/usr/bin' },
  timeoutMs: 30_000,
  maxBufferBytes: 5 * 1024 * 1024,
};
const result = await executeSpawn(input);
// result.stdout / result.stderr / result.exitCode / result.timedOut / result.durationMs
```
