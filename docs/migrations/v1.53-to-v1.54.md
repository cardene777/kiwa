### Pattern 1 — 6 CLI stub 直接呼出

```ts
import { invokeMobileCli } from '@kiwa-lab/mobile';

const r = await invokeMobileCli({
  command: 'expo build',
  args: ['--platform', 'ios'],
  env: { KIWA_MOBILE_MODE: 'real' },
});
console.log(r.stdout);
```

### Pattern 2 — Axis → CLI mapping

```ts
import { cliForAxis, invokeMobileCli, type MobileAxis } from '@kiwa-lab/mobile';

async function runAxisCli(axis: MobileAxis) {
  const cli = cliForAxis(axis);
  if (cli === null) return; // non-CLI-backed axis
  await invokeMobileCli({
    command: cli,
    args: [],
    env: { KIWA_MOBILE_MODE: 'real' },
  });
}
```

### Pattern 3 — Fail-closed 検証

```ts
import { invokeMobileCli } from '@kiwa-lab/mobile';

await expect(
  invokeMobileCli({ command: 'expo build', args: [], env: {} }),
).rejects.toThrow(/KIWA_MOBILE_MODE/);
```
