### Pattern 2 — behavior diff early warning

```ts
import { runFidelityCheck, summarizeFidelityBehaviorDiff } from '@kiwa/desktop';

const diffs = await runFidelityCheck({});
const summary = summarizeFidelityBehaviorDiff(diffs);
// summary.axesWithBehaviorDiff = ['electron', 'clipboard', ...]
// summary.totalMetadataDiffs = N (>0)
```

### Pattern 3 — per-step drill-down

```ts
import { runFidelityCheck } from '@kiwa/desktop';

const diffs = await runFidelityCheck({ axes: ['auto-updater'] });
for (const d of diffs) {
  for (const m of d.metadataDiffs) {
    console.log(`${m.neutralEvent}[${m.stepIndex}] ${m.key}: mock=${m.mockValue} vs real=${m.realValue}`);
  }
}
```
