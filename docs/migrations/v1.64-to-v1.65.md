### Pattern 1 — Snapshot capture

```ts
import { captureSnapshot } from '@kiwa/quality-metrics';

const snapshot = captureSnapshot({
  report: myQualityReport,
  capturedAt: '2026-07-08T00:00:00Z',
  label: 'release-v1.65',
});
```

### Pattern 2 — Regression detection

```ts
import { compareToBaseline, detectDrift } from '@kiwa/quality-metrics';

const comparison = compareToBaseline({ current, baseline });
const drift = detectDrift({ comparison, thresholdPct: 5.0 });

if (drift.category === 'regression') {
  console.log('Blockers:', drift.regressions);
}
```

### Pattern 3 — Trend across releases

```ts
import { generateTrendReport } from '@kiwa/quality-metrics';

const trend = generateTrendReport([snap1, snap2, snap3]);
// trend.axisSummary で axis 別 up/down/flat 判定
```
