### Pattern 1 — subscription 初期化

```ts
import { startLifecycle } from '@kiwa/payment';
const s = startLifecycle({ timestamp: new Date().toISOString() });
```

### Pattern 2 — event 遷移

```ts
import { handleEvent } from '@kiwa/payment';
const next = handleEvent({ session: s, event: 'payment-succeeded', timestamp: t1 });
```

### Pattern 3 — dunning recovery chain

```ts
const grace = handleEvent({ session, event: 'payment-failed', timestamp: t1 });
const dunning = handleEvent({ session: grace, event: 'payment-failed', timestamp: t2 });
const recovered = handleEvent({ session: dunning, event: 'dunning-succeeded', timestamp: t3 });
```

### Pattern 4 — summary 統計

```ts
import { summarizeLifecycle } from '@kiwa/payment';
const summary = summarizeLifecycle(session);
```
