### Pattern 3 — 4 status 経路の handling

```ts
import { probeAndInvoke } from '@kiwa/desktop';

const result = await probeAndInvoke({ axis: 'clipboard', target: 'linux' });
switch (result.status) {
  case 'invoked': /* result.spawnResult は SpawnResult */ break;
  case 'cli-unavailable': /* 未 install、 skip */ break;
  case 'axis-skipped': /* platform mismatch */ break;
  case 'no-cli-mapping': /* semantics-only axis */ break;
}
```
