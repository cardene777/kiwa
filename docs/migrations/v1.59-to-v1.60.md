### Pattern 1 — invokeDesktopCli stub

```ts
import { invokeDesktopCli, type SpawnInvocation } from '@kiwa-test/desktop';

const inv: SpawnInvocation = {
  command: 'ffmpeg',
  args: ['-i', 'input.mp4', 'output.webm'],
  env: { KIWA_DESKTOP_MODE: 'real' },
};
const result = await invokeDesktopCli(inv);
console.log(result.invoked);  // true
console.log(result.stdout);   // "[v0.5 stub] ffmpeg -i input.mp4 output.webm"
```

### Pattern 2 — cliForAxis mapping

```ts
import { cliForAxis } from '@kiwa-test/desktop';

console.log(cliForAxis('auto-updater'));    // 'electron-updater'
console.log(cliForAxis('screen-recording')); // 'ffmpeg'
console.log(cliForAxis('electron'));         // null (non-CLI axis)
```

### Pattern 3 — buildSpawnInvocation factory

```ts
import { buildSpawnInvocation } from '@kiwa-test/desktop';

const inv = buildSpawnInvocation({
  command: 'xclip',
  args: ['-selection', 'clipboard'],
  env: { KIWA_DESKTOP_MODE: 'real' },
  cwd: '/tmp/work',
});
```

### Pattern 4 — env-gate + fail-closed

```ts
import { invokeDesktopCli } from '@kiwa-test/desktop';

// throw = KIWA_DESKTOP_MODE 未設定
await invokeDesktopCli({ command: 'ffmpeg', args: [], env: {} });
// throw = KIWA_DESKTOP_MODE=mock
await invokeDesktopCli({ command: 'ffmpeg', args: [], env: { KIWA_DESKTOP_MODE: 'mock' } });
// throw = args >32
await invokeDesktopCli({ command: 'ffmpeg', args: new Array(33).fill('a'), env: { KIWA_DESKTOP_MODE: 'real' } });
```
