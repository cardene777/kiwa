### Pattern 1 — Fabric renderer

```ts
import { initFabric, scheduleFabricRender, commitShadowTree, completeFabricMount } from '@kiwa/mobile';

const s = initFabric({ target: 'ios', rootId: 'AppRoot' });
scheduleFabricRender(s, 'discrete');
commitShadowTree(s, { nodeCount: 24 });
completeFabricMount(s);
```

### Pattern 2 — TurboModules

```ts
import { initTurboModules, registerTurboSpec, bindJsiRuntime, invokeTurboMethod } from '@kiwa/mobile';

const s = initTurboModules({ target: 'android', moduleName: 'CameraTurbo' });
registerTurboSpec(s, ['takePhoto', 'startRecording']);
bindJsiRuntime(s);
invokeTurboMethod(s, 'takePhoto');
```

### Pattern 3 — Codegen

```ts
import { initCodegen, loadCodegenSchema, generateSpec, emitCodegenType, completeCodegenBuild } from '@kiwa/mobile';

const s = initCodegen({ target: 'ios', packageName: '@myapp/native' });
loadCodegenSchema(s, 'sha256:abc');
generateSpec(s, { specCount: 3 });
emitCodegenType(s, 'NativeCameraSpec.h');
completeCodegenBuild(s);
```

### Pattern 4 — New Architecture full init

```ts
import { initNewArchitecture, startNewArchInit, enableConcurrentReact, bridgeLegacyModule, markNewArchReady } from '@kiwa/mobile';

const s = initNewArchitecture({ target: 'ios', appName: 'MyApp' });
startNewArchInit(s);
enableConcurrentReact(s);
bridgeLegacyModule(s, 'LegacyAudio');
markNewArchReady(s);
```
