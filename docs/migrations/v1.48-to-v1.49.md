### Pattern 1 — React 19 Actions (component)

```ts
import { initializeReactActions, beginActionTransition, resolveAction } from '@kiwa/component';

const s = initializeReactActions({ target: 'playwright-ct', actionId: 'form-1' });
beginActionTransition(s);
resolveAction(s, 'ok');
```

### Pattern 2 — Islands architecture (component)

```ts
import { bootstrapIslandsRoute, registerIsland, beginIslandHydration, markIslandInteractive } from '@kiwa/component';

const s = bootstrapIslandsRoute({ target: 'storybook8', routeId: '/home' });
registerIsland(s, { islandId: 'nav', loadStrategy: 'load', interactiveBoundary: true });
```

### Pattern 3 — Turbopack HMR (nextjs)

```ts
import { startTurbopackHmr, markModuleUpdated, findHmrBoundary, applyHmrPatch, completeFastRefresh } from '@kiwa/nextjs';

const s = startTurbopackHmr({ target: 'app-router', sessionId: 'hmr-1' });
markModuleUpdated(s, 'src/page.tsx');
findHmrBoundary(s, 'src/layout.tsx');
applyHmrPatch(s);
completeFastRefresh(s);
```

### Pattern 4 — Concurrent transitions (nextjs)

```ts
import { startConcurrentTransition, markTransitionPending, interruptTransition, commitTransition } from '@kiwa/nextjs';

const s = startConcurrentTransition({ target: 'app-router', transitionId: 't1' });
markTransitionPending(s);
commitTransition(s, 'v1');
```
