### Pattern 1 — 単一 axis mock adapter

```ts
import { MOCK_ADAPTERS, type AdapterInvocation } from '@kiwa-test/mobile';

const inv: AdapterInvocation = { scanId: '1', target: 'ios', mode: 'mock' };
const r = await MOCK_ADAPTERS.fabric.scan(inv);
```

### Pattern 2 — 全 11 axis 横断 mock scan

```ts
import { MOCK_ADAPTERS, type MobileAxis } from '@kiwa-test/mobile';

const AXES: MobileAxis[] = ['react-native', 'expo', 'metro', 'navigation', 'reanimated', 'async-storage', 'secure-storage', 'fabric', 'turbo-modules', 'codegen', 'new-architecture'];
for (const axis of AXES) {
  const r = await MOCK_ADAPTERS[axis].scan({ scanId: axis, target: 'ios', mode: 'mock' });
  console.log(axis, r.eventCount);
}
```

### Pattern 3 — Fidelity harness

```ts
import { runFidelityCheck, summarizeFidelity } from '@kiwa-test/mobile';

const diffs = await runFidelityCheck(AXES);
const summary = summarizeFidelity(diffs);
console.log(`fidelity: ${summary.matched}/${summary.total}`);
```
