### Pattern 1 — InvokeCache 生成 + withCache 経由 呼出

```ts
import { InvokeCache, withCache } from '@kiwa/desktop';

const cache = new InvokeCache();
const result = await withCache({
  cache,
  invokeInput: { axis: 'clipboard', target: 'macos' },
});
// result.cacheStatus = 'cache-miss' → 2 回目 'cache-hit'
```

### Pattern 2 — LRU + TTL config 上書き

```ts
const strictCache = new InvokeCache({
  ttlMs: 60_000,    // 1 分 TTL
  maxEntries: 32,   // 小さめ LRU
  enabled: true,
});
```

### Pattern 3 — cache-disabled 経路 (backward compat verify)

```ts
const disabled = new InvokeCache({ enabled: false });
// withCache は 毎回 実 probeAndInvoke 実行、 v0.9 相当
```

### Pattern 4 — 手動 invalidate + clear

```ts
const key = buildCacheKey({ axis: 'clipboard', target: 'macos' });
cache.invalidate(key);  // 特定 key 削除
cache.clear();          // 全 clear
```
